8|/** Pure v1 sensor parser. No device authentication, transport ACK or stateful dedupe. */
9|class Esp32PacketDecoder {
10|    private data class NumberToken(val raw: String)
11|    fun decodeSensor(json: String): Esp32SensorPacket? {
12|        if (json.length !in 2..4096) return null
13|        return try {
14|            val fields = mutableMapOf<String, Any?>()
15|            JsonReader(StringReader(json)).use { reader ->
16|                reader.strictness = Strictness.STRICT
17|                reader.beginObject()
18|                while (reader.hasNext()) {
19|                    val key = reader.nextName()
20|                    require(!fields.containsKey(key)) { "Duplicate key" }
21|                    fields[key] = when (reader.peek()) {
22|                        JsonToken.STRING -> reader.nextString()
23|                        JsonToken.NUMBER -> NumberToken(reader.nextString())
24|                        JsonToken.BOOLEAN -> reader.nextBoolean()
25|                        JsonToken.NULL -> { reader.nextNull(); null }
26|                        else -> throw IllegalArgumentException("Primitive fields only")
27|                    }
28|                }
29|                reader.endObject()
30|                require(reader.peek() == JsonToken.END_DOCUMENT)
31|            }
32|            fun integer(name: String): Long {
33|                val raw = (fields[name] as? NumberToken)?.raw ?: error("Required integer")
34|                require(raw.matches(Regex("0|[1-9][0-9]*")))
35|                return raw.toLong()
36|            }
37|            fun percentage(name: String): Int = integer(name).also { require(it in 0..100) }.toInt()
38|            fun decimal(name: String): Float = ((fields[name] as? NumberToken)?.raw?.toFloatOrNull()
39|                ?: error("Required number")).also { require(it.isFinite()) }
40|            fun optional(name: String): Float? = if (fields[name] == null) null else decimal(name)
41|            fun flag(name: String): Boolean = fields[name] as? Boolean ?: error("Required boolean")
42|            val version = integer("protocolVersion"); require(version == 1L)
43|            val id = fields["deviceId"] as? String ?: error("Required ID")
44|            require(id.isNotBlank() && id.length <= 128)
45|            val voltage = if (fields["batteryVoltageMv"] == null) null else integer("batteryVoltageMv").also { require(it <= Int.MAX_VALUE) }.toInt()
46|            Esp32SensorPacket(version.toInt(), id, integer("sequenceNumber"), integer("timestampMs"),
47|                decimal("accelXMs2"), decimal("accelYMs2"), decimal("accelZMs2"),
48|                decimal("gyroXDps"), decimal("gyroYDps"), decimal("gyroZDps"),
49|                optional("pressurePa"), optional("temperatureC"), optional("altitudeDeltaM"),
50|                percentage("batteryPercent"), voltage, flag("isCharging"), flag("sosButtonPressed"), percentage("sensorQuality"))
51|        } catch (_: Exception) {
52|            // Do not retain raw payload or exception text in logs.
53|            null
54|        }
55|    }
56|}

58|    val protocolVersion: Int, val deviceId: String, val sequenceNumber: Long, val timestampMs: Long,
59|    val accelXMs2: Float, val accelYMs2: Float, val accelZMs2: Float,
60|    val gyroXDps: Float, val gyroYDps: Float, val gyroZDps: Float,
61|    val pressurePa: Float?, val temperatureC: Float?, val altitudeDeltaM: Float?,
62|    val batteryPercent: Int, val batteryVoltageMv: Int?, val isCharging: Boolean,
63|    val sosButtonPressed: Boolean, val sensorQuality: Int
64|)
