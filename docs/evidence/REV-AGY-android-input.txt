3|/** Nondecreasing elapsed milliseconds, never civil/wall time. */
4|fun interface MonotonicClock { fun nowMs(): Long }
5|
6|/** Synchronous acceptance: return means SENT; throw Exception means FAILED. */
7|fun interface AlertSink { fun send(alert: Alert) }

9|enum class State { MONITORING, SUSPECTED, VERIFYING, ALERTING, AWAITING_HELP }
10|enum class Response { UNKNOWN, SAFE, NEED_HELP, NO_RESPONSE }
11|enum class Status { NOT_REQUIRED, COUNTDOWN, SENDING, SENT, FAILED, ACKNOWLEDGED }
12|data class Alert(

14|    val response: Response,
15|    val locationMessage: String = "chưa xác định được vị trí"
16|)

19|    val eventId: Long,

26|/**

28| * Host must call tick at the deadline; this core does not schedule background work.
29| * Calls reentering from the synchronous sink cannot finish or resend the active event.
30| */
31|class AlertCore(
32|    private val clock: MonotonicClock,
33|    private val sink: AlertSink,
34|    private val timeoutMs: Long = 10_000,
35|    private val logCapacity: Int = 32
36|) {
37|    init {
38|        require(timeoutMs > 0) { "timeoutMs must be positive" }
39|        require(logCapacity > 0) { "logCapacity must be positive" }
40|    }
41|
42|    private val owner = Thread.currentThread()
43|    private var state = State.MONITORING
44|    private var eventId = 0L
45|    private var response = Response.UNKNOWN
46|    private var status = Status.NOT_REQUIRED
47|    private var started = 0L
48|    private var sending = false
49|    private val log = ArrayDeque<RecordedEvent>()
50|
51|    private fun owned() {
52|        check(Thread.currentThread() === owner) { "Call on owner thread" }
53|    }
54|
55|    /** Display-only read; never sends an alert. */
56|    fun snapshot(): Snapshot {
57|        owned()
58|        val remaining = if (state == State.VERIFYING)
59|            (timeoutMs - (clock.nowMs() - started)).coerceAtLeast(0) else null
60|        return Snapshot(state, eventId, response, status, remaining)
61|    }
62|

64|        owned()

66|    }

71|    }

89|            status = Status.COUNTDOWN
90|            record()
91|        }
92|    }

99|        }

104|        if (state == State.MONITORING) eventId++
105|        response = reason
106|        state = State.ALERTING
107|        status = Status.SENDING
108|        record()

110|        sending = true
111|        try {
112|            sink.send(Alert(eventId, reason))
113|            status = Status.SENT
114|            state = State.AWAITING_HELP
115|        } catch (_: Exception) {
116|            // Never retain transport messages, which may contain private data.
117|            status = Status.FAILED
118|        } finally {
119|            sending = false
120|        }
121|        record()
122|    }
123|
124|    fun tick() {
125|        owned()
126|        if (state == State.VERIFYING && clock.nowMs() - started >= timeoutMs)
127|            dispatch(Response.NO_RESPONSE)
128|    }
129|
130|    fun safe() {
131|        owned()
132|        tick()

134|            response = Response.SAFE
135|            status = Status.NOT_REQUIRED
136|            state = State.MONITORING
137|            record()
138|        }
139|    }

141|    fun needHelp() {
142|        owned()

144|    }
145|
146|    fun sos() {
147|        owned()

149|    }
150|
151|    fun complete() {
152|        owned()
153|        if (!sending && (state == State.AWAITING_HELP || state == State.ALERTING)) {
154|            state = State.MONITORING
155|            status = Status.ACKNOWLEDGED
156|            record()
157|        }
158|    }
